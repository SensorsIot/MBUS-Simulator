# Host tier — FSD §12. Plain compiler, no ESP-IDF, no hardware.
#
#   cmake -S test/host -B build/host -G Ninja
#   cmake --build build/host
#   ctest --test-dir build/host --output-on-failure
#
# The oracle is dlms_parser, an independent DLMS decoder, so a pass here means
# something we did not write can read what the encoder produces. Pinned to a tag,
# not a branch: this exists to notice when the
# library's behaviour changes, which it cannot do if the library moves underneath
# it.
cmake_minimum_required(VERSION 3.22)
project(mbus_simulator_host_tests CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)
FetchContent_Declare(
  dlms_parser
  GIT_REPOSITORY https://github.com/esphome-libs/dlms_parser.git
  GIT_TAG        v2.1.0
  GIT_SHALLOW    TRUE
)
# Consumed via add_subdirectory, which takes dlms_parser's own "used as a
# subproject" branch: just the library, no doctest, no mbedtls, no BearSSL. We
# need none of them — the CII this simulates is not encrypted (FSD A-01).
FetchContent_MakeAvailable(dlms_parser)

# The component under test, built from the same sources the firmware compiles.
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../components/e450_core e450_core)

add_executable(test_encode test_encode.cpp)
target_link_libraries(test_encode PRIVATE e450_core dlms_parser)
target_compile_options(test_encode PRIVATE -Wall -Wextra)

# Development tool, deliberately not a ctest case: decodes a hex dump and prints
# what the oracle made of it. Used to check the harness itself against a real
# meter capture before trusting its verdict on generated bytes.
add_executable(decode_dump decode_dump.cpp)
target_link_libraries(decode_dump PRIVATE dlms_parser)
target_compile_options(decode_dump PRIVATE -Wall -Wextra)

enable_testing()

# One CTest entry per case, so a failure names the property that broke rather
# than "the encode test". The name carries the FSD requirement it discharges.
add_test(NAME FR-ENC-01-04.decodes-independently    COMMAND test_encode decode)
add_test(NAME FR-ENC-05.invoke-id-increments        COMMAND test_encode invoke_id)
add_test(NAME FR-ENC-06-07.gbt-block-structure      COMMAND test_encode gbt)
add_test(NAME FR-ENC-07.block-size-invariant        COMMAND test_encode block_size)
add_test(NAME FR-ENC-08-10.hdlc-framing-and-crc     COMMAND test_encode hdlc)
add_test(NAME FR-MDL-03-05.energy-integrates        COMMAND test_encode energy)
add_test(NAME FR-MDL-04.no-truncation-loss          COMMAND test_encode truncation)
add_test(NAME FR-MDL-07.currents-derived            COMMAND test_encode currents)
add_test(NAME FR-MDL-09.unity-pf-holds-reactive     COMMAND test_encode reactive)
add_test(NAME FR-MDL-10.counters-monotonic          COMMAND test_encode monotonic)
add_test(NAME FR-CLK-04.same-input-same-bytes       COMMAND test_encode determinism)
add_test(NAME FR-SCH-08.telegram-leaves-a-gap       COMMAND test_encode silence)
add_test(NAME FR-MOD-03.simple-mode-one-frame       COMMAND test_encode mode_simple)
add_test(NAME FR-MOD-04.random-mode-varies-repeats  COMMAND test_encode mode_random)

# Phase 2 runtime directives — FSD §14.
add_test(NAME FR-FLT-01.fault-values-are-fixed       COMMAND test_encode fault_fixed)
add_test(NAME FR-FLT-02.fcs-corrupts-one-frame       COMMAND test_encode fault_fcs)
add_test(NAME FR-FLT-03.truncate-leaves-no-flag      COMMAND test_encode fault_truncate)
add_test(NAME FR-FLT-04.drop-block-yields-nothing    COMMAND test_encode fault_drop_block)
add_test(NAME FR-FLT-05.start-mid-frame-recoverable  COMMAND test_encode fault_start_mid)
add_test(NAME FR-FLT-06.duplicate-register-differs   COMMAND test_encode fault_duplicate)
add_test(NAME FR-FLT-07.noise-has-no-framing         COMMAND test_encode fault_noise)
add_test(NAME FR-FLT-02-07.no-fault-is-transparent   COMMAND test_encode fault_none)
add_test(NAME FR-FLT-10-11.identity-and-serial       COMMAND test_encode identity)
add_test(NAME FR-FLT-12.energy-set-exactly           COMMAND test_encode energy)
